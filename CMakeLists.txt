cmake_minimum_required(VERSION 3.16)
project(blake3_provider VERSION 0.1.0 LANGUAGES C)

if(CMAKE_SYSTEM_NAME MATCHES "Linux|FreeBSD|OpenBSD|NetBSD|DragonFly" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  set(_asm_default ON)
else()
  set(_asm_default OFF)
endif()
option(BLAKE3_PROV_USE_ASM "Use upstream .S assembly instead of intrinsics (Linux/BSD x86_64 only)" ${_asm_default})
option(BLAKE3_PROV_MT "Multithreading (reserved; not implemented in v1)" OFF)
option(BLAKE3_PROV_BUILD_TESTS "Build tests/bench (needs -DOPENSSL_CRYPTO_LIBRARY=)" OFF)
set(MODULES_DIR "" CACHE PATH "Install directory for the provider module (default: `openssl version -m`)")
set(BLAKE3_PROV_PREBUILT_MODULE "" CACHE FILEPATH "Use this already-built module for tests instead of building one")

if(BLAKE3_PROV_MT)
  message(FATAL_ERROR "BLAKE3_PROV_MT is a reserved stub: providers must not spawn threads (see README)")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)

list(APPEND CMAKE_MODULE_PATH ${CMAKE_SOURCE_DIR}/cmake)
include(FindOpenSSLHeaders)

set(B3 ${CMAKE_SOURCE_DIR}/third_party/blake3)
set(PROVIDER_SOURCES src/prov_main.c src/prov_digest.c src/prov_mac.c src/prov_kdf.c src/prov_params.c)

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" _proc)
if(CMAKE_C_COMPILER_ARCHITECTURE_ID)
  string(TOLOWER "${CMAKE_C_COMPILER_ARCHITECTURE_ID}" _proc)
endif()
if(_proc MATCHES "^(x86_64|amd64|x64)$")
  set(B3_ARCH x86_64)
elseif(_proc MATCHES "^(i[3-6]86|x86)$")
  set(B3_ARCH x86)
elseif(_proc MATCHES "^(aarch64|arm64|armv8.*)$")
  set(B3_ARCH aarch64)
else()
  set(B3_ARCH other)
endif()
message(STATUS "BLAKE3 provider target arch: ${B3_ARCH}")

set(CORE_SOURCES ${B3}/blake3.c ${B3}/blake3_dispatch.c ${B3}/blake3_portable.c src/b3_dispatch_ext.c)

if(MSVC)
  set(_force_inc /FI${CMAKE_SOURCE_DIR}/src/b3_prefix.h)
else()
  set(_force_inc -include ${CMAKE_SOURCE_DIR}/src/b3_prefix.h)
endif()

if(B3_ARCH STREQUAL x86_64 OR B3_ARCH STREQUAL x86)
  if(BLAKE3_PROV_USE_ASM)
    if(NOT B3_ARCH STREQUAL x86_64 OR MSVC OR APPLE OR WIN32)
      message(FATAL_ERROR "BLAKE3_PROV_USE_ASM is only supported for Linux/BSD x86_64 (symbol prefixing needs the cpp-processed .S files)")
    endif()
    enable_language(ASM)
    list(APPEND CORE_SOURCES ${B3}/blake3_sse2_x86-64_unix.S ${B3}/blake3_sse41_x86-64_unix.S
         ${B3}/blake3_avx2_x86-64_unix.S ${B3}/blake3_avx512_x86-64_unix.S)
    set(B3_BACKEND_KIND asm)
  else()
    list(APPEND CORE_SOURCES ${B3}/blake3_sse2.c ${B3}/blake3_sse41.c ${B3}/blake3_avx2.c ${B3}/blake3_avx512.c)
    if(MSVC)
      set_source_files_properties(${B3}/blake3_avx2.c PROPERTIES COMPILE_OPTIONS /arch:AVX2)
      set_source_files_properties(${B3}/blake3_avx512.c PROPERTIES COMPILE_OPTIONS /arch:AVX512)
    else()
      set_source_files_properties(${B3}/blake3_sse2.c PROPERTIES COMPILE_OPTIONS -msse2)
      set_source_files_properties(${B3}/blake3_sse41.c PROPERTIES COMPILE_OPTIONS -msse4.1)
      set_source_files_properties(${B3}/blake3_avx2.c PROPERTIES COMPILE_OPTIONS -mavx2)
      set_source_files_properties(${B3}/blake3_avx512.c PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512vl")
    endif()
    set(B3_BACKEND_KIND intrinsics)
  endif()
elseif(B3_ARCH STREQUAL aarch64)
  list(APPEND CORE_SOURCES ${B3}/blake3_neon.c)
  set(B3_BACKEND_KIND intrinsics)
else()
  set(B3_BACKEND_KIND portable)
endif()
message(STATUS "BLAKE3 SIMD sources: ${B3_BACKEND_KIND}")

add_library(b3core OBJECT ${CORE_SOURCES})
target_include_directories(b3core PRIVATE ${B3} src)
target_compile_options(b3core PRIVATE ${_force_inc})
target_compile_definitions(b3core PRIVATE BLAKE3_TESTING)
if(MSVC)
  target_compile_definitions(b3core PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(b3core PRIVATE /O2)
else()
  target_compile_options(b3core PRIVATE -O3 -fvisibility=hidden)
endif()
set_property(TARGET b3core PROPERTY ASM_FLAGS "-include ${CMAKE_SOURCE_DIR}/src/b3_prefix.h")

if(BLAKE3_PROV_BUILD_TESTS)
  set(STOCK_SOURCES ${CORE_SOURCES})
  list(REMOVE_ITEM STOCK_SOURCES src/b3_dispatch_ext.c)
  add_library(b3stock STATIC ${STOCK_SOURCES})
  target_include_directories(b3stock PRIVATE ${B3})
  target_compile_definitions(b3stock PRIVATE BLAKE3_TESTING)
  if(NOT MSVC)
    target_compile_options(b3stock PRIVATE -O3)
  endif()
endif()

if(NOT BLAKE3_PROV_PREBUILT_MODULE)
  add_library(blake3 MODULE ${PROVIDER_SOURCES} $<TARGET_OBJECTS:b3core>)
  target_include_directories(blake3 PRIVATE ${OPENSSL_INCLUDE_DIR} ${B3} src)
  target_compile_options(blake3 PRIVATE ${_force_inc})
  target_compile_options(blake3 PRIVATE $<$<NOT:$<C_COMPILER_ID:MSVC>>:-Wall -Wextra -O3 -fvisibility=hidden>
                                        $<$<C_COMPILER_ID:MSVC>:/O2 /W4>)
  target_compile_definitions(blake3 PRIVATE $<$<C_COMPILER_ID:MSVC>:_CRT_SECURE_NO_WARNINGS>)
  set_target_properties(blake3 PROPERTIES PREFIX "" OUTPUT_NAME blake3)
  if(APPLE)
    set_target_properties(blake3 PROPERTIES SUFFIX ".dylib")
    target_link_options(blake3 PRIVATE -Wl,-exported_symbol,_OSSL_provider_init)
  elseif(WIN32)
    target_sources(blake3 PRIVATE ${CMAKE_SOURCE_DIR}/src/blake3.def)
  else()
    target_link_options(blake3 PRIVATE -Wl,--version-script=${CMAKE_SOURCE_DIR}/src/exports.map -Wl,-z,defs)
  endif()

  if(NOT MODULES_DIR)
    find_program(_openssl_bin openssl)
    if(_openssl_bin)
      execute_process(COMMAND ${_openssl_bin} version -m OUTPUT_VARIABLE _mods OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
      string(REGEX MATCH "\"([^\"]+)\"" _m "${_mods}")
      if(CMAKE_MATCH_1)
        set(MODULES_DIR "${CMAKE_MATCH_1}")
      endif()
    endif()
  endif()
  if(MODULES_DIR)
    install(TARGETS blake3 LIBRARY DESTINATION ${MODULES_DIR} RUNTIME DESTINATION ${MODULES_DIR})
  else()
    message(STATUS "MODULES_DIR not set and no openssl binary found: 'cmake --install' will be skipped")
  endif()
  install(FILES conf/openssl-blake3.cnf.example README.md DESTINATION share/doc/blake3-provider)
endif()

if(BLAKE3_PROV_BUILD_TESTS)
  enable_testing()
  add_subdirectory(test)
  add_subdirectory(bench)
endif()
