set(OPENSSL_CRYPTO_LIBRARY "" CACHE FILEPATH "libcrypto to link the tests against")
if(NOT OPENSSL_CRYPTO_LIBRARY)
  message(FATAL_ERROR "tests need -DOPENSSL_CRYPTO_LIBRARY=<path to libcrypto>")
endif()
if(BLAKE3_PROV_PREBUILT_MODULE)
  get_filename_component(B3_MODDIR "${BLAKE3_PROV_PREBUILT_MODULE}" DIRECTORY)
else()
  set(B3_MODDIR $<TARGET_FILE_DIR:blake3>)
endif()
find_package(Threads)

set(T_ENV "B3_MODULE_DIR=${B3_MODDIR}" "B3_VECTORS=${CMAKE_CURRENT_SOURCE_DIR}/vectors/test_vectors.json")
foreach(t spec_traces test_vectors test_incremental test_evp test_threads)
  add_executable(${t} ${t}.c ref_blake3.c)
  target_include_directories(${t} PRIVATE ${OPENSSL_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(${t} PRIVATE ${OPENSSL_CRYPTO_LIBRARY} Threads::Threads)
  if(NOT MSVC)
    target_compile_options(${t} PRIVATE -O1 -g)
  endif()
endforeach()

foreach(impl auto portable sse2 sse41 avx2 avx512 neon)
  foreach(t spec_traces test_vectors test_incremental test_evp)
    add_test(NAME ${t}.${impl} COMMAND ${t})
    set(_e ${T_ENV})
    if(NOT impl STREQUAL auto)
      list(APPEND _e "BLAKE3_PROV_FORCE_IMPL=${impl}")
    endif()
    set_tests_properties(${t}.${impl} PROPERTIES ENVIRONMENT "${_e}" SKIP_RETURN_CODE 77)
  endforeach()
endforeach()
add_test(NAME test_threads COMMAND test_threads)
set_tests_properties(test_threads PROPERTIES ENVIRONMENT "${T_ENV}" SKIP_RETURN_CODE 77)

add_executable(eval_provider cross/eval_provider.c ref_blake3.c)
target_include_directories(eval_provider PRIVATE ${OPENSSL_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(eval_provider PRIVATE ${OPENSSL_CRYPTO_LIBRARY})

add_executable(fuzz_ops fuzz/fuzz_ops.c ref_blake3.c)
target_include_directories(fuzz_ops PRIVATE ${OPENSSL_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(fuzz_ops PRIVATE ${OPENSSL_CRYPTO_LIBRARY})
foreach(impl auto portable sse2 sse41 avx2 avx512)
  add_test(NAME fuzz_quick.${impl} COMMAND fuzz_ops 2 1)
  set(_e ${T_ENV})
  if(NOT impl STREQUAL auto)
    list(APPEND _e "BLAKE3_PROV_FORCE_IMPL=${impl}")
  endif()
  set_tests_properties(fuzz_quick.${impl} PROPERTIES ENVIRONMENT "${_e}" SKIP_RETURN_CODE 77)
endforeach()
if(CMAKE_C_COMPILER_ID MATCHES "Clang" AND NOT APPLE)   # Apple clang ships no libclang_rt.fuzzer
  add_executable(fuzz_ops_libfuzzer fuzz/fuzz_ops.c ref_blake3.c)
  target_compile_definitions(fuzz_ops_libfuzzer PRIVATE FUZZ_LIBFUZZER)
  target_include_directories(fuzz_ops_libfuzzer PRIVATE ${OPENSSL_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
  target_compile_options(fuzz_ops_libfuzzer PRIVATE -fsanitize=fuzzer,address -g)
  target_link_options(fuzz_ops_libfuzzer PRIVATE -fsanitize=fuzzer,address)
  target_link_libraries(fuzz_ops_libfuzzer PRIVATE ${OPENSSL_CRYPTO_LIBRARY})
endif()
